cmake_minimum_required(VERSION 3.20)
project(face_track_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Maxine bridge currently requires Windows x64")
endif()
set(MAXINE_SOURCE "" CACHE PATH "NVIDIA Maxine-AR-SDK source checkout")
find_path(AR_INCLUDE nvAR.h HINTS "${MAXINE_SOURCE}" PATH_SUFFIXES nvAR/include include REQUIRED)
find_path(CV_INCLUDE nvCVImage.h HINTS "${MAXINE_SOURCE}" PATH_SUFFIXES nvCVImage/include nvCVImage include REQUIRED)
find_package(CUDAToolkit REQUIRED)
add_library(face_track_bridge SHARED bridge.cpp)
target_compile_features(face_track_bridge PRIVATE cxx_std_17)
target_include_directories(face_track_bridge PRIVATE "${AR_INCLUDE}" "${CV_INCLUDE}" "${CUDAToolkit_INCLUDE_DIRS}")
target_compile_definitions(face_track_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(face_track_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(face_track_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
endforeach()
